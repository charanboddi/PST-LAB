#include <bits/stdc++.h>
using namespace std;

long long stringSimilarity(string s) {
    int n = s.length();

    vector<int> z(n);
    int left = 0, right = 0;

    z[0] = n;

    for (int i = 1; i < n; i++) {

        if (i <= right) {
            z[i] = min(right - i + 1, z[i - left]);
        }

        while (i + z[i] < n &&
               s[z[i]] == s[i + z[i]]) {
            z[i]++;
        }

        if (i + z[i] - 1 > right) {
            left = i;
            right = i + z[i] - 1;
        }
    }

    long long answer = 0;

    for (int i = 0; i < n; i++) {
        answer += z[i];
    }

    return answer;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        string s;
        cin >> s;

        cout << stringSimilarity(s) << '\n';
    }

    return 0;
}
